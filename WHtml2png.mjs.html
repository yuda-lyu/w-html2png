//getBundledChrome, 取得套件自帶免安裝chrome之執行檔路徑: 經ensureBins取得檔案組chrome(首次使用時下載或自鏡像取得並校驗, 之後自快取取得亦校驗), 回傳組內portable/App/Chrome-bin/<chrome版本>/chrome.exe
//chrome版本由清單(bin/manifest.json)之組內路徑決定, 組內須恰有一個, 否則reject帶code為E_MANIFEST之Error
let getBundledChrome = async () => {
    let r = await ensureBins()
    let files = r.bundles.chrome.files
    let ks = Object.keys(files).filter((k) => /^portable\/App\/Chrome-bin\/[^/]+\/chrome\.exe$/.test(k))
    if (ks.length !== 1) {
        let err = new Error(`the bundle chrome must have exactly one portable/App/Chrome-bin/<version>/chrome.exe, found ${ks.length}`)
        err.code = 'E_MANIFEST'
        throw err
    }
    return files[ks[0]]
}

//raceLimit, 限時等待pm(Promise或回傳Promise之函數, 函數於計時器建立後才呼叫, 使其同步前置耗時亦計入), 超過ms回傳vTimeout; 計時器於結果確定後一律清除, 不留計時器使Node行程可立即退出
let raceLimit = async (pm, ms, vTimeout) => {
    let timer = null
    try {
        let pmTimer = new Promise((resolve) => {
            timer = setTimeout(() => {
                resolve(vTimeout)
            }, ms)
        })
        let p = isfun(pm) ? pm() : pm
        return await Promise.race([p, pmTimer])
    }
    finally {
        clearTimeout(timer)
    }
}


//trackPage, 追蹤頁面之load與網路活動, 須於載入前呼叫; st.tActive為最後一次活動時刻, st.nActive為活動累計數, st.inflight為進行中之請求, st.onIdle於進行中之請求清空或頁面關閉時呼叫
let trackPage = (page) => {
    let st = {
        tActive: 0,
        nActive: 0,
        inflight: new Set(),
        onIdle: null,
    }
    let active = () => {
        st.tActive = Date.now()
        st.nActive++
    }
    let idle = () => {
        if (st.onIdle) {
            st.onIdle()
        }
    }
    let done = (r) => {
        st.inflight.delete(r)
        active()
        if (st.inflight.size === 0) {
            idle()
        }
    }
    page.on('load', active)
    page.on('request', (r) => {
        st.inflight.add(r)
        active()
    })
    page.on('requestfinished', done)
    page.on('requestfailed', done)
    page.on('close', idle)
    return st
}


//waitPageSettled, 等待頁面確實完成才可檢查頁面錯誤或截圖, 回傳結果字串, 不拋錯(未達成者續往下, 由其後之步驟自然反應):
//'settled'已靜止; 'no-load'限時內未load或頁面已關閉; 'busy'限時內未能靜止(持續有請求); 'no-timer'頁面計時器未回應(主執行緒卡住或setTimeout被頁面取代); 'error'頁面內等待反覆失敗(如持續導覽)
//playwright之networkidle只是Node端之500ms計時器(無請求進行中即起算), 與頁面執行進度無關: 頁面解析或執行腳本耗時超過500ms(負載高、腳本大、同步運算)時會早於頁尾腳本與load, 此時頁面錯誤尚未發生、內容尚未繪製, 故須另行保證:
//1. load: 解析完成、同步腳本與load事件處理皆已執行, 其間之頁面錯誤與load事件循同一連線依序送達, load送達時錯誤必已收到
//2. 靜止窗: 自load與最後一次網路活動起timeSettle內無請求, 且以頁面自身之計時器等待: 頁面計時器依到期時間排序, 故load前所設且延遲不超過timeSettle之計時器必先執行(不論主執行緒多忙), 其引發之請求亦被追蹤, 未處理之rejection亦已回報; 頁面內等待之回應與頁面錯誤亦循同一連線依序送達
let waitPageSettled = async (page, st) => {
    let tEnd = Date.now() + timeSettleMax
    let left = () => tEnd - Date.now()
    let nError = 0
    while (true) {

        //load, 導覽至新文件時等待新文件之load
        let loaded = await Promise.resolve()
            .then(() => page.waitForLoadState('load', { timeout: Math.max(left(), 1) }))
            .then(() => true, () => false)
        if (!loaded) {
            return 'no-load'
        }

        //inflight, 等待進行中之請求結束
        let idle = await raceLimit(new Promise((resolve) => {
            st.onIdle = () => {
                resolve(true)
            }
            if (st.inflight.size === 0) {
                resolve(true)
            }
        }), left(), false)
        st.onIdle = null
        if (!idle) {
            return 'busy'
        }

        //window, 以頁面自身之計時器等到靜止窗結束; 已過亦至少往返頁面一次(取1ms), 使已到期之計時器與已排入之回報先執行
        let nActive = st.nActive
        let ms = Math.min(Math.max(timeSettle - (Date.now() - st.tActive), 1), timeSettle)
        let r = await raceLimit(
            Promise.resolve().then(() => page.evaluate((v) => new Promise((resolve) => {
                setTimeout(resolve, v)
            }), ms)).then(() => 'ok', () => 'error'),
            left(),
            'no-timer',
        )
        if (r === 'no-timer') {
            return r
        }
        if (r === 'error') {

            //導覽使執行環境銷毀時重新等待新文件, 其他原因(如頁面已關閉)於下一輪等待load時結束; 限次數免持續導覽或頁面內持續拋錯時空轉
            nError++
            if (nError >= 5) {
                return r
            }
            continue
        }

        //check, 窗內無新活動即靜止, 否則再等一個窗
        if (st.nActive === nActive) {
            return 'settled'
        }
        if (left() <= 0) {
            return 'busy'
        }

    }
}


//isPid, 是否為可用之行程pid(安全正整數)

}


//isConnected, 回傳瀏覽器是否仍與本行程連線: true/false, 無法判定(isConnected拋錯)回傳null
let isConnected = (browser) => {
    try {
        return browser.isConnected() === true
    }
    catch (err) {
        return null
    }
}


//getBrowserPid, 經本瀏覽器之CDP連線取得其主程序pid; 須於瀏覽器仍可回應時取得(卡死後CDP已不回應), 逾timeProbe或失敗回傳null, 不拋錯
//pid出自本實例所啟動之瀏覽器自身的連線, 為其後強制結束時確認歸屬之依據
let getBrowserPid = async (browser) => {
    let core = async () => {
        let cdp = await browser.newBrowserCDPSession()
        try {
            let { processInfo } = await cdp.send('SystemInfo.getProcessInfo')
            let r = processInfo.find((v) => v.type === 'browser')
            return r && isPid(r.id) ? r.id : null
        }
        finally {
            try {
                cdp.detach().catch(() => {})
            }
            catch (err) {}
        }
    }
    return raceLimit(core().catch(() => null), timeProbe, null)
}


//waitDisconnected, 等待瀏覽器斷線事件(行程結束後playwright發出disconnected), 限時內未斷線回傳false
let waitDisconnected = async (browser) => {
    let off = () => {}
    let pm = new Promise((resolve) => {
        let fn = () => {
            resolve(true)
        }
        try {
            browser.on('disconnected', fn)
            off = () => {
                browser.off('disconnected', fn)
            }
        }
        catch (err) {}
        if (isConnected(browser) === false) {
            resolve(true)
        }
    })
    let r = await raceLimit(pm, timeProbe, false)
    try {
        off()
    }
    catch (err) {}
    return r
}


//killBrowser, 確認pid仍是本實例所啟動之瀏覽器才樹狀強制結束(wsemi execProcessKillPid, Windows為taskkill /T /F, 不用child.kill()), 回傳結果字串:
//'killed'已結束且確認斷線; 'gone'瀏覽器已斷線(與本行程之連線已斷, 無須也不可再殺); 'no-pid'pid未取得(無法確認歸屬)不殺; 'unverified'連線狀態無法判定不殺; 'failed'結束指令失敗或逾時; 'unconfirmed'已下結束指令但限時內未見斷線
//歸屬確認: pid取自本瀏覽器自身之CDP連線, 且本瀏覽器仍連線中(playwright於行程結束時才發disconnected, 行程仍在時作業系統不會將該pid配給他人; 已斷線則不殺, 避免pid被重用時誤殺他人行程)
//pid可為數字或回傳數字之函數(結束當下才取值, 使關閉等待期間才查得之pid仍可用)
let killBrowser = async (browser, pidOrFun, trace) => {
    let result = ''
    let pid = isfun(pidOrFun) ? pidOrFun() : pidOrFun
    let connected = isConnected(browser)
    if (!isPid(pid)) {
        result = 'no-pid'
    }
    else if (connected === false) {
        result = 'gone'
    }
    else if (connected === null) {
        result = 'unverified'
    }
    else {
        let ok = await raceLimit(execProcessKillPid(pid).then(() => true, () => false), timeProbe, false)
        if (ok) {
            result = (await waitDisconnected(browser)) ? 'killed' : 'unconfirmed'
        }
        else {
            result = isConnected(browser) === false ? 'gone' : 'failed'
        }
    }
    if (trace) {
        trace('browser-kill', { result, pid: isPid(pid) ? pid : null })
    }
    return result
}


//closeBrowser, 限時關閉瀏覽器: timeClose內未關閉(主程序卡死)則經killBrowser強制結束; 回傳'closed'或killBrowser之結果字串
//browser.close()成功或(拋錯但已斷線)視為已關閉; 拋錯而仍連線者不是已關閉, 立即走killBrowser
let closeBrowser = async (browser, pid, trace) => {
    if (trace) {
        trace('browser-close-start', {})
    }
    let closed = await raceLimit(
        Promise.resolve().then(() => browser.close()).then(() => true, () => isConnected(browser) === false),
        timeClose,
        false,
    )
    if (trace) {
        trace('browser-close', { closed })
    }
    if (closed) {
        return 'closed'
    }
    return killBrowser(browser, pid, trace)
}


//ensurePid, 取得世代之瀏覽器主程序pid(取得後每世代共用), 回傳pid或null; 取不到時不永久快取, 下一個給timeout之呼叫再試一次(每世代最多2次, 免瀏覽器持續無回應時每次呼叫都多等timeProbe)
let ensurePid = (g, browser) => {
    if (g.pmPid === undefined) {
        g.pidTries++
        g.pmPid = getBrowserPid(browser)
            .then((pid) => {
                g.pid = pid
                if (pid === null && g.pidTries < 2) {
                    g.pmPid = undefined
                }
                return pid
            })
    }
    return g.pmPid
}


//getGen, 取得現行世代, 無則lazy啟動; 併發呼叫共用同一個launch Promise
let getGen = (executablePath, modeHeadless) => {

    if (gen === null) {

        //g, 同步指派gen, 同一tick後之併發呼叫直接取得同一世代, 不會重複launch
        let g = {
            id: ++genSeq,
            pmBrowser: null,
            ctxPms: {},
            browser: undefined,
            guarded: false,
            pmPid: undefined,
            pidTries: 0,
            pid: undefined,
            pmClose: undefined,
        }
        g.pmBrowser = chromium.launch(getLaunchOpt(executablePath, modeHeadless))
            .then((browser) => {

                //browser
                g.browser = browser

                //disconnected, 瀏覽器意外死亡時, 若仍為現行世代則歸零, 使下次呼叫重新啟動
                browser.on('disconnected', () => {
                    if (gen === g) {
                        gen = null
                    }
                })

                return browser
            })
            .catch((err) => {

                //啟動失敗, 若仍為現行世代則歸零, 使下次呼叫可重試啟動
                if (gen === g) {
                    gen = null
                }

                return Promise.reject(err)
            })
        gen = g

    }

    return gen
}


//closeGen, 關閉指定世代; 若其為現行世代先歸零, 使後續呼叫直接啟動新世代, 不會取得關閉中之瀏覽器
//曾有呼叫給opt.timeout之世代(guarded)才限時關閉並可強制結束, 回傳'closed'或killBrowser之結果字串, 啟動中(尚無瀏覽器可關)回傳'pending'並於啟動完成後在背景關閉;
//未曾給opt.timeout之世代維持原行為(不限時等待啟動與關閉), 回傳null
let closeGen = async (g, trace = null) => {
    if (!g) {
        return null
    }
    if (gen === g) {
        gen = null
    }
    if (g.guarded !== true) {
        await g.pmBrowser
            .then((browser) => {
                return browser.close()
            })
            .catch(() => {})
        return null
    }
    if (g.pmClose === undefined) {
        if (g.browser === undefined) {

            //尚在啟動: 不等待啟動完成(啟動期間無從取得pid), 完成後取pid再於背景關閉(背景之處置經trace通知, 可能在呼叫已返回之後)
            g.pmClose = g.pmBrowser
                .then(async (browser) => {
                    await ensurePid(g, browser)
                    return closeBrowser(browser, () => g.pid, trace)
                })
                .catch(() => 'failed')
            return 'pending'
        }

        //同一世代之關閉(各呼叫之逾時處置、重試前自癒、close()、閒置關閉)共用同一次處置, 不重複關閉或結束
        g.pmClose = closeBrowser(g.browser, () => g.pid, trace)
    }
    else if (g.browser === undefined) {
        return 'pending'
    }
    return g.pmClose
}


//getContext, 取得指定deviceScaleFactor之常駐context, playwright之deviceScaleFactor為context層級故依scale分快取; 快取綁定世代, 等待啟動期間世代已被關閉則拋錯; guard為true時另取瀏覽器主程序pid
let getContext = async (g, scale, guard) => {

    //browser
    let browser = await g.pmBrowser
    if (gen !== g) {
        throw new Error('resident browser has been closed')
    }

    //pid, 給了opt.timeout之呼叫才取(經CDP, 每世代共用), 供逾時卡死時確認歸屬後強制結束
    if (guard) {
        await ensurePid(g, browser)
        if (gen !== g) {
            throw new Error('resident browser has been closed')
        }
    }

    //key
    let key = `dsf-${scale}`

    //newContext
    if (g.ctxPms[key] === undefined) {
        let pmCtx = newContext(browser, scale)
            .catch((err) => {

                //建立失敗清除快取, 使下次呼叫可重試建立
                if (g.ctxPms[key] === pmCtx) {
                    delete g.ctxPms[key]
                }

                return Promise.reject(err)
            })
        g.ctxPms[key] = pmCtx
    }

    let context = null
    try {
        context = await g.ctxPms[key]
    }
    catch (err) {
        //瀏覽器已啟動但無法建立context, 屬瀏覽器層錯誤, 附上世代供重試前關閉本世代
        let e = new Error(errToMsg(err))
        e.whGen = g
        throw e
    }

    return {
        context,
        g,
    }
}


//scheduleIdleClose, 重設閒置計時器, 閒置超過timeIdle且無繪圖中呼叫時自動關閉常駐瀏覽器
let scheduleIdleClose = () => {

    //clear
    clearTimeout(timerIdle)
    timerIdle = null

    //check, 無瀏覽器則無須排程
    if (gen === null) {
        return
    }

    //setTimeout
    timerIdle = setTimeout(() => {
        timerIdle = null
        if (refCount === 0 && gen !== null) {
            closeGen(gen)
        }
    }, timeIdle)

}


//aliveStrategy, 常駐模式: 使用模組層常駐瀏覽器與context, 出圖後不釋放; 只有瀏覽器層錯誤才於重試前關閉本呼叫所用之世代
let aliveStrategy = {
    begin: () => {
        return {
            epoch: closeEpoch,
        }
    },
    isAborted: (st) => {
        return st.epoch !== closeEpoch
    },
    acquire: async ({ executablePath, modeHeadless, scale, guard, onToken }) => {

        //g, 先同步取得本呼叫所等待之世代並回報, 使逾時時關閉的是本呼叫所用者, 而非當下之現行世代
        let g = getGen(executablePath, modeHeadless)
        onToken(g)
        if (guard) {
            g.guarded = true
        }

        let { context } = await getContext(g, scale, guard)
        return {
            context,
            release: null, //常駐不釋放, 瀏覽器交由閒置計時器或WHtml2png.close()關閉
            token: g,
        }
    },
    heal: async (token, trace) => {
        return closeGen(token, trace)
    },
}


//singleStrategy, 單次模式: 每次啟動新瀏覽器, 出圖後即關閉釋放
let singleStrategy = {
    begin: () => {
        return {}
    },
    isAborted: () => {
        return false
    },
    acquire: async ({ executablePath, modeHeadless, scale, guard, trace, onRelease }) => {

        //browser
        let browser = await chromium.launch(getLaunchOpt(executablePath, modeHeadless))

        //pid, 給了opt.timeout之呼叫才取(經CDP), 供逾時卡死時確認歸屬後強制結束; 結束當下才取值(取pid期間逾時亦可用)
        let pid = null

        //pmRelease, 釋放只執行一次: 逾時處置與出圖流程收尾可能各自呼叫, 共用同一次關閉; 給了opt.timeout者以closeBrowser限時關閉
        let pmRelease = null
        let release = () => {
            if (pmRelease === null) {
                if (guard) {
                    pmRelease = closeBrowser(browser, () => pid, trace)
                }
                else {
                    pmRelease = browser.close()
                }
            }
            return pmRelease
        }

        //onRelease, 啟動完成即回報釋放函數, 使其後取pid或建立context卡住而逾時時, 逾時處置仍能關閉本瀏覽器
        onRelease(release)

        if (guard) {
            pid = await getBrowserPid(browser)
        }

        //context
        let context = null
        try {
            context = await newContext(browser, scale)
        }
        catch (err) {
            //建立context失敗須關閉已啟動之瀏覽器避免洩漏
            await release()
                .catch(() => {})
            throw err
        }

        return {
            context,
            release,
            token: null,
        }
    },
    heal: async () => {
        return null
    },
}


//core, alive與single共用之出圖流程: 驗證輸入, 組合html, 依strategy取得context, 開page載入渲染截圖, 失敗重試, 回傳base64; reject一律為錯誤訊息字串
async function core(width, height, scale, html, opt, strategy) {

    //st, 本次呼叫開始時之狀態(alive用於判斷其後是否被WHtml2png.close())
    let st = strategy.begin()

    //executablePath, 可由外部指定瀏覽器執行檔(如本機安裝Chrome或playwright託管chromium), 未指定則使用套件自帶之免安裝chrome
    let executablePath = get(opt, 'executablePath')
    if (!isestr(executablePath)) {
        executablePath = ''
    }

    //isWindows, 套件自帶之免安裝chrome只有Windows版, 故未指定executablePath時須為Windows
    if (!isWindows() && executablePath === '') {
        return Promise.reject('operating system is not windows, the bundled portable chrome supports windows only, please give opt.executablePath')
    }

    //width
    if (!isnum(width)) {
        return Promise.reject('width is not a number')
    }
    width = cint(width)
    if (width <= 0) {
        return Promise.reject('width <= 0')
    }

    //height
    if (!isnum(height)) {
        return Promise.reject('height is not a number')
    }
    height = cint(height)
    if (height <= 0) {
        return Promise.reject('height <= 0')
    }

    //scale
    if (!isnum(scale)) {
        return Promise.reject('scale is not a number')
    }
    scale = cdbl(scale)
    if (scale <= 0) {
        return Promise.reject('scale <= 0')
    }

    //html
    if (!isestr(html)) {
        return Promise.reject('html is not an effective string')
    }

    //modeHeadless
    let modeHeadless = get(opt, 'modeHeadless')
    if (modeHeadless !== true && modeHeadless !== false && modeHeadless !== 'new' && modeHeadless !== 'shell') {
        modeHeadless = 'new' //無頭, 不顯示UI
        // modeHeadless = false //顯示UI
    }

    //stylesHead
    let stylesHead = get(opt, 'stylesHead')
    if (!isearr(stylesHead)) {
        stylesHead = []
    }

    //scriptsHead
    let scriptsHead = get(opt, 'scriptsHead')
    if (!isearr(scriptsHead)) {
        scriptsHead = []
    }

    //execJsHead
    let execJsHead = get(opt, 'execJsHead')
    if (isestr(execJsHead)) {
        execJsHead = [execJsHead]
    }
    if (!isearr(execJsHead)) {
        execJsHead = []
    }

    //execJsPost
    let execJsPost = get(opt, 'execJsPost')
    if (isestr(execJsPost)) {
        execJsPost = [execJsPost]
    }
    if (!isearr(execJsPost)) {
        execJsPost = []
    }

    //funGetUrl
    let funGetUrl = get(opt, 'funGetUrl')

    //funPageWait
    let funPageWait = get(opt, 'funPageWait')

    //rejectOnPageError, 頁面端未捕捉之例外(pageerror)是否視為失敗
    let rejectOnPageError = get(opt, 'rejectOnPageError') === true

    //executablePath
    if (executablePath !== '') {

        //check
        if (!isFile(executablePath)) {
            return Promise.reject(`invalid opt.executablePath[${executablePath}]`)
        }

    }
    else {

        //getBundledChrome, 經ensureBins取得套件自帶之免安裝chrome(檔案組chrome), 取不到時reject錯誤訊息字串'[錯誤碼] 說明'(錯誤碼如E_OFFLINE_MISSING、E_HASH_MISMATCH)
        let r = await getBundledChrome()
            .then((fp) => {
                return { fp }
            }, (err) => {
                return { err }
            })
        if (r.err !== undefined) {
            return Promise.reject(`[${get(r.err, 'code', 'E_ENSURE_BINS')}] ${errToMsg(r.err)}`)
        }
        executablePath = r.fp

    }

    //retry, 總嘗試次數(含第一次)
    let retry = get(opt, 'retry')
    if (!ispint(retry)) {
        retry = 3
    }
    retry = cint(retry)

    //timeout, 單次出圖(每次嘗試)之允許時長(ms), 未給或非預期之輸入(非有效正數、超過計時器上限2147483647、讀取拋錯)視為未給, 不限時(null)
    let timeout = null
    try {
        let v = get(opt, 'timeout')
        if (isenum(v, { use: 'positive', max: 2147483647 })) {
            timeout = Math.ceil(Number(v)) //有效數字字串轉為數值後使用; 計時器以整數毫秒為粒度(Node截斷小數), 向上取整使不早於所給時長
        }
    }
    catch (err) {
        timeout = null
    }

    //trace, 觀察回調(opt.onTrace), 未給時為null; 回調屬觀察用途, 不得改變本函數既有行為
    let trace = null
    try {
        trace = getTrace(opt)
    }
    catch (err) {
        trace = null
    }

    //writeError
    let writeError = get(opt, 'writeError')
    if (!isbol(writeError)) {
        writeError = false
    }

    //fdHtml
    let fdHtml = get(opt, 'fdHtml')
    if (!isestr(fdHtml)) {
        fdHtml = './_convertTemp'
    }

    //fdErr
    let fdErr = get(opt, 'fdErr')
    if (!isestr(fdErr)) {
        fdErr = './_convertTemp'
    }

    //g, 組合頁面html(單次掃描模板, 內容逐字填入)
    let g = buildHtml({
        html,
        stylesHead,
        scriptsHead,
        execJsHead,
        execJsPost,
    })

    //idpm
    let idpm = `${now2strp()}-${genID(6)}`

    //iCore
    let iCore = 0

    //writeErrFile, 寫錯誤檔, 失敗回傳錯誤訊息(不拋錯, 不覆蓋原錯誤)
    let writeErrFile = (fn, c) => {
        try {
            fs.mkdirSync(fdErr, { recursive: true })
            fs.writeFileSync(path.resolve(fdErr, fn), c, 'utf8')
            return ''
        }
        catch (err) {
            return errToMsg(err)
        }
    }

    //removeHtml, 刪除funGetUrl場景之html暫存檔(只刪自建者), 失敗時依writeError寫錯誤檔, 不拋錯
    let removeHtml = (fpHtml, id) => {
        try {
            if (isFile(fpHtml)) {
                fs.unlinkSync(fpHtml)
            }
        }
        catch (err) {
            if (writeError) {
                writeErrFile(`err_${id}_fpHtml.json`, `can not delete html[whweb_${id}.html]: ${errToMsg(err)}`)
            }
        }
    }

    //newProgress, 建立一次嘗試之進度紀錄: step為進行中之步驟(逾時時據以回報卡在哪一步), page/token/release為已取得者, useBrowser為已開始取用瀏覽器, cancelled為已被逾時處置放棄(其進行中之步驟自行收尾)
    let newProgress = () => {
        return {
            attempt: 0,
            step: 'start',
            page: null,
            token: null,
            release: null,
            fpHtml: null,
            useBrowser: false,
            cancelled: false,
        }
    }

    //setStep, 記錄進行中之步驟
    let setStep = (c, step) => {
        c.step = step
        if (trace) {
            trace('step', { attempt: c.attempt, step })
        }
    }

    //checkCancelled, 已被逾時處置放棄之嘗試不再往下操作頁面(拋錯由exec之fun(page)攔截)
    let checkCancelled = (c) => {
        if (c.cancelled) {
            if (trace) {
                trace('cancelled', { attempt: c.attempt, step: c.step })
            }
            throw new Error('cancelled by timeout')
        }
    }

    //exec, 執行一次出圖, 回傳結果物件{ state, message, b64, needHeal, token }, 不拋錯
    let exec = async(c) => {

        //iCore
        iCore++

        //id
        let id = `${idpm}-${iCore}`

        //attempt
        c.attempt = iCore
        if (trace) {
            trace('attempt', { attempt: c.attempt, timeout })
        }

        let earrs = []
        let b64 = ''
        let fpHtml = null
        let needHeal = false
        let token = null

        try {

            //check, 本呼叫開始後已被WHtml2png.close()則不再啟動瀏覽器
            if (strategy.isAborted(st)) {
                earrs.push({
                    anchor: 'WHtml2png.close()',
                    err: 'aborted by WHtml2png.close()',
                })
            }

            //url, 僅funGetUrl場景需寫html暫存檔, 否則直接以setContent灌入html字串
            let url = null
            if (size(earrs) === 0 && isfun(funGetUrl)) {

                //writeFileSync
                try {
                    setStep(c, 'write html file for funGetUrl')
                    fs.mkdirSync(fdHtml, { recursive: true })
                    let fp = path.resolve(fdHtml, `whweb_${id}.html`)
                    fs.writeFileSync(fp, g, { encoding: 'utf8', flag: 'wx' }) //排他建立, 不覆寫既有檔案
                    fpHtml = fp
                    c.fpHtml = fp
                }
                catch (err) {
                    earrs.push({
                        anchor: 'write html file for funGetUrl',
                        err,
                    })
                }

                //funGetUrl
                if (size(earrs) === 0) {
                    try {
                        setStep(c, 'funGetUrl')
                        url = funGetUrl(fpHtml)
                        if (ispm(url)) {
                            url = await url
                        }
                    }
                    catch (err) {
                        earrs.push({
                            anchor: 'funGetUrl(fpHtml)',
                            err,
                        })
                    }
                }

            }

            //page流程
            if (size(earrs) === 0) {

                //acquire, 依strategy取得context與釋放函數; 等待funGetUrl期間若已被WHtml2png.close()則不再啟動瀏覽器
                let context = null
                let release = null
                if (strategy.isAborted(st)) {
                    earrs.push({
                        anchor: 'WHtml2png.close()',
                        err: 'aborted by WHtml2png.close()',
                    })
                }
                else if (c.cancelled) {

                    //cancelled, 等待funGetUrl期間已逾時被放棄, 不再取得瀏覽器
                    earrs.push({
                        anchor: 'funGetUrl(fpHtml)',
                        err: 'cancelled by timeout',
                    })
                    if (trace) {
                        trace('cancelled', { attempt: c.attempt, step: 'funGetUrl' })
                    }

                }
                else {
                    try {
                        setStep(c, 'strategy.acquire')
                        c.useBrowser = true
                        let ac = await strategy.acquire({
                            executablePath,
                            modeHeadless,
                            scale,
                            guard: timeout !== null,
                            trace,
                            onToken: (t) => {
                                c.token = t
                            },
                            onRelease: (f) => {
                                c.release = f

                                //已逾時被放棄(逾時處置時瀏覽器尚在啟動, 無釋放函數可用)者, 瀏覽器一啟動完成即釋放, 不論其後取pid或建context是否卡住
                                if (c.cancelled) {
                                    f().catch(() => {})
                                }
                            },
                        })
                        context = ac.context
                        release = ac.release
                        token = ac.token
                        c.release = release
                        c.token = token

                        //cancelled, 取得期間已逾時被放棄, 取得之瀏覽器不再使用, 立即釋放(alive之世代已由逾時處置關閉)
                        if (c.cancelled) {
                            context = null
                            earrs.push({
                                anchor: 'strategy.acquire',
                                err: 'cancelled by timeout',
                            })
                            if (trace) {
                                trace('cancelled', { attempt: c.attempt, step: 'strategy.acquire' })
                            }
                        }
                    }
                    catch (err) {
                        if (err && err.whGen) {
                            needHeal = true
                            token = err.whGen
                        }
                        earrs.push({
                            anchor: 'strategy.acquire',
                            err,
                        })
                    }
                }

                //page
                let page = null
                if (context !== null) {
                    try {
                        setStep(c, 'context.newPage()')
                        page = await context.newPage()

                        //cancelled, 開頁期間已逾時被放棄, 不再出圖, 關閉該頁面(不等待, 免被已卡住之瀏覽器拖住收尾)
                        if (c.cancelled) {
                            page.close()
                                .catch(() => {})
                            page = null
                            earrs.push({
                                anchor: 'context.newPage()',
                                err: 'cancelled by timeout',
                            })
                            if (trace) {
                                trace('cancelled', { attempt: c.attempt, step: 'context.newPage()' })
                            }
                        }
                        else {
                            c.page = page
                            page.setDefaultNavigationTimeout(60 * 1000) //延長timeout
                        }
                    }
                    catch (err) {
                        needHeal = true //無法開頁屬瀏覽器層錯誤
                        earrs.push({
                            anchor: 'context.newPage()',
                            err,
                        })
                    }
                }

                //draw
                if (page !== null) {

                    //pageerror, 須於載入前掛上
                    let pageErrors = []
                    if (rejectOnPageError) {
                        page.on('pageerror', (e) => {
                            pageErrors.push(errToMsg(e))
                        })
                    }

                    //stPage, 追蹤load與網路活動, 須於載入前掛上
                    let stPage = trackPage(page)

                    try {

                        //setViewportSize, deviceScaleFactor已由context給定
                        setStep(c, 'page.setViewportSize()')
                        await page.setViewportSize({
                            width: Number(width),
                            height: Number(height),
                        })
                        checkCancelled(c)

                        //show page, networkidle代表500ms內無網路連線, 對應puppeteer之networkidle系列
                        if (url !== null) {
                            setStep(c, 'page.goto()')
                            await page.goto(url, {
                                waitUntil: 'networkidle',
                                timeout: 60 * 1000, //延長timeout
                            })
                        }
                        else {
                            setStep(c, 'page.setContent()')
                            await page.setContent(g, {
                                waitUntil: 'networkidle',
                                timeout: 60 * 1000, //延長timeout
                            })
                        }
                        checkCancelled(c)

                        //waitPageSettled, networkidle可能早於頁尾腳本與load, 須等頁面確實完成才可檢查頁面錯誤或截圖
                        setStep(c, 'waitPageSettled')
                        let settled = await waitPageSettled(page, stPage)
                        if (trace) {
                            trace('page-settled', { attempt: c.attempt, result: settled })
                        }
                        checkCancelled(c)

                        //delay
                        if (isfun(funPageWait)) {
                            setStep(c, 'funPageWait')
                            await page.waitForFunction(funPageWait, null, { polling: 200, timeout: 60 * 1000 }) //200ms偵測一次, 延長timeout
                            checkCancelled(c)
                        }

                        //check pageerror, 截圖前有頁面端例外則視為失敗
                        if (rejectOnPageError && pageErrors.length > 0) {
                            let e = new Error(`page error: ${pageErrors.join(' | ')}`)
                            e.isPageError = true
                            throw e
                        }

                        //screenshot, 直接取Buffer, 不落地png暫存檔
                        setStep(c, 'page.screenshot()')
                        let buf = await page.screenshot({
                            timeout: 60 * 1000, //延長timeout
                        })

                        //b64
                        b64 = Buffer.from(buf).toString('base64')

                    }
                    catch (err) {
                        earrs.push({
                            anchor: 'fun(page)',
                            err,
                        })

                        //載入/等待/截圖失敗時一併附上已收到之頁面端錯誤(原因保留)
                        if (rejectOnPageError && pageErrors.length > 0 && !(err && err.isPageError)) {
                            earrs.push({
                                anchor: 'pageerror',
                                err: `page error: ${pageErrors.join(' | ')}`,
                            })
                        }
                    }

                    //close
                    try {
                        setStep(c, 'page.close()')
                        await page.close()
                    }
                    catch (err) {
                        earrs.push({
                            anchor: 'page.close()',
                            err,
                        })
                    }

                    //check, 瀏覽器已斷線屬瀏覽器層錯誤
                    try {
                        if (!context.browser().isConnected()) {
                            needHeal = true
                        }
                    }
                    catch (err) {}

                }

                //release, single模式關閉本次啟動之瀏覽器, alive模式為null不釋放
                if (isfun(release)) {
                    try {
                        setStep(c, 'release()')
                        await release()
                    }
                    catch (err) {
                        earrs.push({
                            anchor: 'release()',
                            err,
                        })
                    }
                }

            }

        }
        catch (err) {
            earrs.push({
                anchor: 'exec',
                err,
            })
        }

        //delete fpHtml
        if (fpHtml !== null) {
            removeHtml(fpHtml, id)
        }

        //check
        if (size(earrs) > 0) {

            //cearrs
            let cearrs = ''
            each(earrs, (earr) => {
                cearrs += `${errToMsg(earr.err)} in ${earr.anchor}\n`
            })

            //writeErrFile, 寫檔失敗時附記於訊息, 不覆蓋原錯誤; 已被逾時處置放棄者, 錯誤檔由逾時處置寫入(同檔名), 此處不覆蓋
            if (writeError && !c.cancelled) {
                let e = writeErrFile(`err_${id}_all.json`, cearrs)
                if (e !== '') {
                    cearrs += `can not write error file: ${e}\n`
                }
            }

            return {
                state: 'error',
                message: cearrs,
                needHeal,
                token,
            }
        }

        //check
        if (!isestr(b64)) {
            return {
                state: 'error',
                message: 'b64 is not an effective string',
                needHeal,
                token,
            }
        }

        return {
            state: 'success',
            b64,
        }
    }

    //notes, 逾時處置之瀏覽器處置結果對應之訊息(括號內), 未列者(含頁面層之'closed')不附註
    let notes = {
        'killed': 'browser force-killed',
        'no-pid': 'browser not killed, pid unknown',
        'unverified': 'browser not killed, connection state unknown',
        'failed': 'browser kill failed',
        'unconfirmed': 'browser kill not confirmed',
        'pending': 'browser will be closed after launch',
    }

    //execLimited, 限制一次出圖(每次嘗試)之允許時長(opt.timeout); 未給即直接執行exec, 流程與未支援前相同
    //逾時不等exec結束(其進行中之步驟於頁面或瀏覽器關閉後自行收尾, c.cancelled使其不再往下取用瀏覽器), 依卡住之層級處置, 無論是否再重試皆於此完成:
    //已開頁面且timeProbe內關得掉: 頁面層問題, 瀏覽器仍可回應, 不重啟(不牽連並發呼叫, 符合R12);
    //尚無頁面(取得瀏覽器或開頁卡住)或頁面關不掉: 瀏覽器層問題, 關閉本呼叫所用之瀏覽器(alive為本呼叫所用世代, 下次呼叫即啟動新世代; single為本呼叫專屬之瀏覽器), 主程序卡死時timeClose後確認歸屬再樹狀強制結束;
    //尚未取用瀏覽器(如funGetUrl卡住)之逾時不動瀏覽器
    let execLimited = async (c) => {

        //check
        if (timeout === null) {
            return exec(c)
        }

        //race
        let r = await raceLimit(() => exec(c), timeout, null)
        if (r !== null) {
            return r
        }

        //cancelled
        c.cancelled = true
        let step = c.step
        if (trace) {
            trace('timeout', { attempt: c.attempt, step, timeout })
        }

        //pageClosed, 關得掉頁面代表瀏覽器仍可回應, 屬頁面層問題
        let pageClosed = false
        if (c.page !== null) {
            if (trace) {
                trace('page-close-start', { attempt: c.attempt })
            }
            pageClosed = await raceLimit(
                Promise.resolve().then(() => c.page.close()).then(() => true, () => false),
                timeProbe,
                false,
            )
            if (trace) {
                trace('page-close', { attempt: c.attempt, closed: pageClosed })
            }
        }

        //browserLevel
        let browserLevel = c.useBrowser && !pageClosed

        //release, single模式為本呼叫專屬之瀏覽器, 不論層級皆釋放(與exec收尾共用同一次關閉)
        let outcome = null
        if (isfun(c.release)) {
            outcome = await Promise.resolve().then(() => c.release()).catch(() => 'failed')
        }

        //heal, alive模式之瀏覽器層問題關閉本呼叫所用之世代
        if (browserLevel && c.token !== null) {
            let o = await strategy.heal(c.token, trace)
            if (o !== null) {
                outcome = o
            }
        }

        //delete fpHtml, funGetUrl卡住時exec可能永不結束, 暫存檔由此刪除
        if (c.fpHtml !== null) {
            removeHtml(c.fpHtml, `${idpm}-${c.attempt}`)
        }

        //message
        let message = `timeout[${timeout}ms] in ${step}`
        let note = notes[outcome]
        if (!isestr(note) && browserLevel) {
            if (outcome === 'closed') {
                note = 'browser closed'
            }
            else if (outcome === 'gone') {
                note = 'browser disconnected'
            }
        }
        if (isestr(note)) {
            message += ` (${note})`
        }

        //writeErrFile
        if (writeError) {
            let e = writeErrFile(`err_${idpm}-${c.attempt}_all.json`, `${message}\n`)
            if (e !== '') {
                message += `\ncan not write error file: ${e}`
            }
        }

        return {
            state: 'error',
            message,
            needHeal: false, //已於上處置
            token: null,
        }
    }

    //proc, 失敗時重試
    let errTemp = null
    let b64 = ''
    while (true) {

        //exec
        let r = await execLimited(newProgress())
            .catch((err) => {
                //exec已全攔截, 預期不會來到此處, 仍保守轉為錯誤結果
                return {
                    state: 'error',
                    message: errToMsg(err),
                    needHeal: false,
                    token: null,
                }
            })

        //check, r.state='success'
        if (r.state === 'success') {
            b64 = r.b64
            break
        }

        //check, 若retry=3, 執行3次都失敗時iCore=3, 則不再重試; 本呼叫開始後已被WHtml2png.close()亦不再重試
        if (iCore >= retry || strategy.isAborted(st)) {
            errTemp = r.message
            break
        }

        //自癒, 只有瀏覽器層錯誤才關閉本呼叫所用之世代(alive), 其他並發呼叫之瀏覽器不受頁面層錯誤影響
        if (r.needHeal) {
            await strategy.heal(r.token, trace)
        }

        //延遲再重試
        if (trace) {
            trace('retry', { attempt: iCore, wait: 5000 })
        }
        await delay(5000)

    }

    //check
    if (errTemp !== null) {
        return Promise.reject(errToMsg(errTemp))
    }

    return b64
}


//coreAlive, 常駐模式: 出圖後瀏覽器留存供同行程後續呼叫重用, 含引用計數, 結束時重設閒置關閉排程
async function coreAlive(width, height, scale, html, opt = {}) {
    refCount++
    try {
        let b64 = await core(width, height, scale, html, opt, aliveStrategy)
        return b64
    }
    finally {
        refCount--
        scheduleIdleClose()
    }
}


//coreSingle, 單次模式: 每次啟動新瀏覽器出圖後即關閉釋放, Node行程可自然結束
async function coreSingle(width, height, scale, html, opt = {}) {
    let b64 = await core(width, height, scale, html, opt, singleStrategy)
    return b64
}

async function WHtml2png(width = 700, height = 400, scale = 3, html = '', opt = {}) {
    // console.log('WHtml2png', width, height, scale, opt)

    //mode
    let mode = get(opt, 'mode')
    if (mode !== 'single' && mode !== 'alive') {
        mode = 'single'
    }

    //dispatch
    if (mode === 'alive') {
        return coreAlive(width, height, scale, html, opt)
            .catch((err) => {
                return Promise.reject(errToMsg(err))
            })
    }
    return coreSingle(width, height, scale, html, opt)
        .catch((err) => {
            return Promise.reject(errToMsg(err))
        })
}
