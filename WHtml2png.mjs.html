//timeIdle, alive模式閒置自動關閉常駐瀏覽器之時間(ms)
let timeIdle = 60 * 60 * 1000 //1hr


//常駐瀏覽器狀態(模組層單例), 供alive模式使用
//gen為現行世代{ id, pmBrowser, ctxPms }: pmBrowser為chromium.launch之Promise, ctxPms為以deviceScaleFactor為key之context Promise快取; null代表下次呼叫須啟動新世代
let gen = null
let genSeq = 0
let closeEpoch = 0 //每次WHtml2png.close()遞增, 其前開始之呼叫不再重試重啟瀏覽器
let refCount = 0 //繪圖中之呼叫數
let timerIdle = null //閒置計時器


function isWindows() {
    return process.platform === 'win32'
}


//isFile, 判斷是否為檔案(跟隨符號連結)
let isFile = (fp) => {
    try {
        return fs.statSync(fp).isFile()
    }
    catch (err) {
        return false
    }
}


//getBundledChrome, 取得套件自帶免安裝chrome之執行檔路徑: 經ensureBins取得檔案組chrome(首次使用時下載或自鏡像取得並校驗, 之後自快取取得亦校驗), 回傳組內portable/App/Chrome-bin/<chrome版本>/chrome.exe
//chrome版本由清單(bin/manifest.json)之組內路徑決定, 組內須恰有一個, 否則reject帶code為E_MANIFEST之Error
let getBundledChrome = async () => {
    let r = await ensureBins()
    let files = r.bundles.chrome.files
    let ks = Object.keys(files).filter((k) => /^portable\/App\/Chrome-bin\/[^/]+\/chrome\.exe$/.test(k))
    if (ks.length !== 1) {
        let err = new Error(`the bundle chrome must have exactly one portable/App/Chrome-bin/<version>/chrome.exe, found ${ks.length}`)
        err.code = 'E_MANIFEST'
        throw err
    }
    return files[ks[0]]
}


//getLaunchOpt, 組合chromium啟動參數, alive與single模式共用
let getLaunchOpt = (executablePath, modeHeadless) => {

    //headless, playwright僅支援布林, modeHeadless=false時顯示UI, 其餘視為無頭
    let headless = modeHeadless !== false

    //launchOpt
    let launchOpt = {
        headless,
        timeout: 60 * 1000, //延長launch timeout
        args: [
            '--lang=en-US', //未指定font-family時預設字型依瀏覽器UI語系解析, zh-TW系統會變成微軟正黑(無襯線), 固定en-US使其與puppeteer時代輸出一致(Times New Roman襯線)
            '--no-sandbox',
            '--disable-gpu',
            '--disable-dev-shm-usage',
            '--disable-sync',
            '--disable-extensions',
            '--disable-default-apps',
            '--metrics-recording-only',
            '--mute-audio',
            '--no-first-run',
            '--safebrowsing-disable-auto-update',
        ],
    }
    if (isestr(executablePath)) {
        launchOpt.executablePath = executablePath
    }

    return launchOpt
}


//newContext, 建立context並攔截模板之函式庫網址改由本機供應, 失敗時關閉已建立之context
let newContext = async (browser, scale) => {
    let context = await browser.newContext({
        viewport: {
            width: 1280,
            height: 720,
        },
        deviceScaleFactor: scale,
    })
    try {
        await routeLibs(context)
    }
    catch (err) {
        await context.close()
            .catch(() => {})
        throw err
    }
    return context
}


//getGen, 取得現行世代, 無則lazy啟動; 併發呼叫共用同一個launch Promise
let getGen = (executablePath, modeHeadless) => {

    if (gen === null) {

        //g, 同步指派gen, 同一tick後之併發呼叫直接取得同一世代, 不會重複launch
        let g = {
            id: ++genSeq,
            pmBrowser: null,
            ctxPms: {},
        }
        g.pmBrowser = chromium.launch(getLaunchOpt(executablePath, modeHeadless))
            .then((browser) => {

                //disconnected, 瀏覽器意外死亡時, 若仍為現行世代則歸零, 使下次呼叫重新啟動
                browser.on('disconnected', () => {
                    if (gen === g) {
                        gen = null
                    }
                })

                return browser
            })
            .catch((err) => {

                //啟動失敗, 若仍為現行世代則歸零, 使下次呼叫可重試啟動
                if (gen === g) {
                    gen = null
                }

                return Promise.reject(err)
            })
        gen = g

    }

    return gen
}


//closeGen, 關閉指定世代; 若其為現行世代先歸零, 使後續呼叫直接啟動新世代, 不會取得關閉中之瀏覽器
let closeGen = async (g) => {
    if (!g) {
        return
    }
    if (gen === g) {
        gen = null
    }
    await g.pmBrowser
        .then((browser) => {
            return browser.close()
        })
        .catch(() => {})
}


//getContext, 取得指定deviceScaleFactor之常駐context, playwright之deviceScaleFactor為context層級故依scale分快取; 快取綁定世代, 等待啟動期間世代已被關閉則拋錯
let getContext = async (executablePath, modeHeadless, scale) => {

    //g
    let g = getGen(executablePath, modeHeadless)

    //browser
    let browser = await g.pmBrowser
    if (gen !== g) {
        throw new Error('resident browser has been closed')
    }

    //key
    let key = `dsf-${scale}`

    //newContext
    if (g.ctxPms[key] === undefined) {
        let pmCtx = newContext(browser, scale)
            .catch((err) => {

                //建立失敗清除快取, 使下次呼叫可重試建立
                if (g.ctxPms[key] === pmCtx) {
                    delete g.ctxPms[key]
                }

                return Promise.reject(err)
            })
        g.ctxPms[key] = pmCtx
    }

    let context = null
    try {
        context = await g.ctxPms[key]
    }
    catch (err) {
        //瀏覽器已啟動但無法建立context, 屬瀏覽器層錯誤, 附上世代供重試前關閉本世代
        let e = new Error(errToMsg(err))
        e.whGen = g
        throw e
    }

    return {
        context,
        g,
    }
}


//scheduleIdleClose, 重設閒置計時器, 閒置超過timeIdle且無繪圖中呼叫時自動關閉常駐瀏覽器
let scheduleIdleClose = () => {

    //clear
    clearTimeout(timerIdle)
    timerIdle = null

    //check, 無瀏覽器則無須排程
    if (gen === null) {
        return
    }

    //setTimeout
    timerIdle = setTimeout(() => {
        timerIdle = null
        if (refCount === 0 && gen !== null) {
            closeGen(gen)
        }
    }, timeIdle)

}


//aliveStrategy, 常駐模式: 使用模組層常駐瀏覽器與context, 出圖後不釋放; 只有瀏覽器層錯誤才於重試前關閉本呼叫所用之世代
let aliveStrategy = {
    begin: () => {
        return {
            epoch: closeEpoch,
        }
    },
    isAborted: (st) => {
        return st.epoch !== closeEpoch
    },
    acquire: async ({ executablePath, modeHeadless, scale }) => {
        let { context, g } = await getContext(executablePath, modeHeadless, scale)
        return {
            context,
            release: null, //常駐不釋放, 瀏覽器交由閒置計時器或WHtml2png.close()關閉
            token: g,
        }
    },
    heal: async (token) => {
        await closeGen(token)
    },
}


//singleStrategy, 單次模式: 每次啟動新瀏覽器, 出圖後即關閉釋放
let singleStrategy = {
    begin: () => {
        return {}
    },
    isAborted: () => {
        return false
    },
    acquire: async ({ executablePath, modeHeadless, scale }) => {

        //browser
        let browser = await chromium.launch(getLaunchOpt(executablePath, modeHeadless))

        //context
        let context = null
        try {
            context = await newContext(browser, scale)
        }
        catch (err) {
            //建立context失敗須關閉已啟動之瀏覽器避免洩漏
            await browser.close()
                .catch(() => {})
            throw err
        }

        return {
            context,
            release: async () => {
                await browser.close()
            },
            token: null,
        }
    },
    heal: async () => {},
}


//core, alive與single共用之出圖流程: 驗證輸入, 組合html, 依strategy取得context, 開page載入渲染截圖, 失敗重試, 回傳base64; reject一律為錯誤訊息字串
async function core(width, height, scale, html, opt, strategy) {

    //st, 本次呼叫開始時之狀態(alive用於判斷其後是否被WHtml2png.close())
    let st = strategy.begin()

    //executablePath, 可由外部指定瀏覽器執行檔(如本機安裝Chrome或playwright託管chromium), 未指定則使用套件自帶之免安裝chrome
    let executablePath = get(opt, 'executablePath')
    if (!isestr(executablePath)) {
        executablePath = ''
    }

    //isWindows, 套件自帶之免安裝chrome只有Windows版, 故未指定executablePath時須為Windows
    if (!isWindows() && executablePath === '') {
        return Promise.reject('operating system is not windows, the bundled portable chrome supports windows only, please give opt.executablePath')
    }

    //width
    if (!isnum(width)) {
        return Promise.reject('width is not a number')
    }
    width = cint(width)
    if (width <= 0) {
        return Promise.reject('width <= 0')
    }

    //height
    if (!isnum(height)) {
        return Promise.reject('height is not a number')
    }
    height = cint(height)
    if (height <= 0) {
        return Promise.reject('height <= 0')
    }

    //scale
    if (!isnum(scale)) {
        return Promise.reject('scale is not a number')
    }
    scale = cdbl(scale)
    if (scale <= 0) {
        return Promise.reject('scale <= 0')
    }

    //html
    if (!isestr(html)) {
        return Promise.reject('html is not an effective string')
    }

    //modeHeadless
    let modeHeadless = get(opt, 'modeHeadless')
    if (modeHeadless !== true && modeHeadless !== false && modeHeadless !== 'new' && modeHeadless !== 'shell') {
        modeHeadless = 'new' //無頭, 不顯示UI
        // modeHeadless = false //顯示UI
    }

    //stylesHead
    let stylesHead = get(opt, 'stylesHead')
    if (!isearr(stylesHead)) {
        stylesHead = []
    }

    //scriptsHead
    let scriptsHead = get(opt, 'scriptsHead')
    if (!isearr(scriptsHead)) {
        scriptsHead = []
    }

    //execJsHead
    let execJsHead = get(opt, 'execJsHead')
    if (isestr(execJsHead)) {
        execJsHead = [execJsHead]
    }
    if (!isearr(execJsHead)) {
        execJsHead = []
    }

    //execJsPost
    let execJsPost = get(opt, 'execJsPost')
    if (isestr(execJsPost)) {
        execJsPost = [execJsPost]
    }
    if (!isearr(execJsPost)) {
        execJsPost = []
    }

    //funGetUrl
    let funGetUrl = get(opt, 'funGetUrl')

    //funPageWait
    let funPageWait = get(opt, 'funPageWait')

    //rejectOnPageError, 頁面端未捕捉之例外(pageerror)是否視為失敗
    let rejectOnPageError = get(opt, 'rejectOnPageError') === true

    //executablePath
    if (executablePath !== '') {

        //check
        if (!isFile(executablePath)) {
            return Promise.reject(`invalid opt.executablePath[${executablePath}]`)
        }

    }
    else {

        //getBundledChrome, 經ensureBins取得套件自帶之免安裝chrome(檔案組chrome), 取不到時reject錯誤訊息字串'[錯誤碼] 說明'(錯誤碼如E_OFFLINE_MISSING、E_HASH_MISMATCH)
        let r = await getBundledChrome()
            .then((fp) => {
                return { fp }
            }, (err) => {
                return { err }
            })
        if (r.err !== undefined) {
            return Promise.reject(`[${get(r.err, 'code', 'E_ENSURE_BINS')}] ${errToMsg(r.err)}`)
        }
        executablePath = r.fp

    }

    //retry, 總嘗試次數(含第一次)
    let retry = get(opt, 'retry')
    if (!ispint(retry)) {
        retry = 3
    }
    retry = cint(retry)

    //writeError
    let writeError = get(opt, 'writeError')
    if (!isbol(writeError)) {
        writeError = false
    }

    //fdHtml
    let fdHtml = get(opt, 'fdHtml')
    if (!isestr(fdHtml)) {
        fdHtml = './_convertTemp'
    }

    //fdErr
    let fdErr = get(opt, 'fdErr')
    if (!isestr(fdErr)) {
        fdErr = './_convertTemp'
    }

    //g, 組合頁面html(單次掃描模板, 內容逐字填入)
    let g = buildHtml({
        html,
        stylesHead,
        scriptsHead,
        execJsHead,
        execJsPost,
    })

    //idpm
    let idpm = `${now2strp()}-${genID(6)}`

    //iCore
    let iCore = 0

    //writeErrFile, 寫錯誤檔, 失敗回傳錯誤訊息(不拋錯, 不覆蓋原錯誤)
    let writeErrFile = (fn, c) => {
        try {
            fs.mkdirSync(fdErr, { recursive: true })
            fs.writeFileSync(path.resolve(fdErr, fn), c, 'utf8')
            return ''
        }
        catch (err) {
            return errToMsg(err)
        }
    }

    //exec, 執行一次出圖, 回傳結果物件{ state, message, b64, needHeal, token }, 不拋錯
    let exec = async() => {

        //iCore
        iCore++

        //id
        let id = `${idpm}-${iCore}`

        let earrs = []
        let b64 = ''
        let fpHtml = null
        let needHeal = false
        let token = null

        try {

            //check, 本呼叫開始後已被WHtml2png.close()則不再啟動瀏覽器
            if (strategy.isAborted(st)) {
                earrs.push({
                    anchor: 'WHtml2png.close()',
                    err: 'aborted by WHtml2png.close()',
                })
            }

            //url, 僅funGetUrl場景需寫html暫存檔, 否則直接以setContent灌入html字串
            let url = null
            if (size(earrs) === 0 && isfun(funGetUrl)) {

                //writeFileSync
                try {
                    fs.mkdirSync(fdHtml, { recursive: true })
                    let fp = path.resolve(fdHtml, `whweb_${id}.html`)
                    fs.writeFileSync(fp, g, { encoding: 'utf8', flag: 'wx' }) //排他建立, 不覆寫既有檔案
                    fpHtml = fp
                }
                catch (err) {
                    earrs.push({
                        anchor: 'write html file for funGetUrl',
                        err,
                    })
                }

                //funGetUrl
                if (size(earrs) === 0) {
                    try {
                        url = funGetUrl(fpHtml)
                        if (ispm(url)) {
                            url = await url
                        }
                    }
                    catch (err) {
                        earrs.push({
                            anchor: 'funGetUrl(fpHtml)',
                            err,
                        })
                    }
                }

            }

            //page流程
            if (size(earrs) === 0) {

                //acquire, 依strategy取得context與釋放函數; 等待funGetUrl期間若已被WHtml2png.close()則不再啟動瀏覽器
                let context = null
                let release = null
                if (strategy.isAborted(st)) {
                    earrs.push({
                        anchor: 'WHtml2png.close()',
                        err: 'aborted by WHtml2png.close()',
                    })
                }
                else {
                    try {
                        let ac = await strategy.acquire({ executablePath, modeHeadless, scale })
                        context = ac.context
                        release = ac.release
                        token = ac.token
                    }
                    catch (err) {
                        if (err && err.whGen) {
                            needHeal = true
                            token = err.whGen
                        }
                        earrs.push({
                            anchor: 'strategy.acquire',
                            err,
                        })
                    }
                }

                //page
                let page = null
                if (context !== null) {
                    try {
                        page = await context.newPage()
                        page.setDefaultNavigationTimeout(60 * 1000) //延長timeout
                    }
                    catch (err) {
                        needHeal = true //無法開頁屬瀏覽器層錯誤
                        earrs.push({
                            anchor: 'context.newPage()',
                            err,
                        })
                    }
                }

                //draw
                if (page !== null) {

                    //pageerror, 須於載入前掛上
                    let pageErrors = []
                    if (rejectOnPageError) {
                        page.on('pageerror', (e) => {
                            pageErrors.push(errToMsg(e))
                        })
                    }

                    try {

                        //setViewportSize, deviceScaleFactor已由context給定
                        await page.setViewportSize({
                            width: Number(width),
                            height: Number(height),
                        })

                        //show page, networkidle代表500ms內無網路連線, 對應puppeteer之networkidle系列
                        if (url !== null) {
                            await page.goto(url, {
                                waitUntil: 'networkidle',
                                timeout: 60 * 1000, //延長timeout
                            })
                        }
                        else {
                            await page.setContent(g, {
                                waitUntil: 'networkidle',
                                timeout: 60 * 1000, //延長timeout
                            })
                        }

                        //delay
                        if (isfun(funPageWait)) {
                            await page.waitForFunction(funPageWait, null, { polling: 200, timeout: 60 * 1000 }) //200ms偵測一次, 延長timeout
                        }

                        //check pageerror, 截圖前有頁面端例外則視為失敗
                        if (rejectOnPageError && pageErrors.length > 0) {
                            let e = new Error(`page error: ${pageErrors.join(' | ')}`)
                            e.isPageError = true
                            throw e
                        }

                        //screenshot, 直接取Buffer, 不落地png暫存檔
                        let buf = await page.screenshot({
                            timeout: 60 * 1000, //延長timeout
                        })

                        //b64
                        b64 = Buffer.from(buf).toString('base64')

                    }
                    catch (err) {
                        earrs.push({
                            anchor: 'fun(page)',
                            err,
                        })

                        //載入/等待/截圖失敗時一併附上已收到之頁面端錯誤(原因保留)
                        if (rejectOnPageError && pageErrors.length > 0 && !(err && err.isPageError)) {
                            earrs.push({
                                anchor: 'pageerror',
                                err: `page error: ${pageErrors.join(' | ')}`,
                            })
                        }
                    }

                    //close
                    try {
                        await page.close()
                    }
                    catch (err) {
                        earrs.push({
                            anchor: 'page.close()',
                            err,
                        })
                    }

                    //check, 瀏覽器已斷線屬瀏覽器層錯誤
                    try {
                        if (!context.browser().isConnected()) {
                            needHeal = true
                        }
                    }
                    catch (err) {}

                }

                //release, single模式關閉本次啟動之瀏覽器, alive模式為null不釋放
                if (isfun(release)) {
                    try {
                        await release()
                    }
                    catch (err) {
                        earrs.push({
                            anchor: 'release()',
                            err,
                        })
                    }
                }

            }

        }
        catch (err) {
            earrs.push({
                anchor: 'exec',
                err,
            })
        }

        //delete fpHtml
        if (fpHtml !== null) {
            try {
                if (isFile(fpHtml)) {
                    fs.unlinkSync(fpHtml)
                }
            }
            catch (err) {
                if (writeError) {
                    writeErrFile(`err_${id}_fpHtml.json`, `can not delete html[whweb_${id}.html]: ${errToMsg(err)}`)
                }
            }
        }

        //check
        if (size(earrs) > 0) {

            //cearrs
            let cearrs = ''
            each(earrs, (earr) => {
                cearrs += `${errToMsg(earr.err)} in ${earr.anchor}\n`
            })

            //writeErrFile, 寫檔失敗時附記於訊息, 不覆蓋原錯誤
            if (writeError) {
                let e = writeErrFile(`err_${id}_all.json`, cearrs)
                if (e !== '') {
                    cearrs += `can not write error file: ${e}\n`
                }
            }

            return {
                state: 'error',
                message: cearrs,
                needHeal,
                token,
            }
        }

        //check
        if (!isestr(b64)) {
            return {
                state: 'error',
                message: 'b64 is not an effective string',
                needHeal,
                token,
            }
        }

        return {
            state: 'success',
            b64,
        }
    }

    //proc, 失敗時重試
    let errTemp = null
    let b64 = ''
    while (true) {

        //exec
        let r = await exec()
            .catch((err) => {
                //exec已全攔截, 預期不會來到此處, 仍保守轉為錯誤結果
                return {
                    state: 'error',
                    message: errToMsg(err),
                    needHeal: false,
                    token: null,
                }
            })

        //check, r.state='success'
        if (r.state === 'success') {
            b64 = r.b64
            break
        }

        //check, 若retry=3, 執行3次都失敗時iCore=3, 則不再重試; 本呼叫開始後已被WHtml2png.close()亦不再重試
        if (iCore >= retry || strategy.isAborted(st)) {
            errTemp = r.message
            break
        }

        //自癒, 只有瀏覽器層錯誤才關閉本呼叫所用之世代(alive), 其他並發呼叫之瀏覽器不受頁面層錯誤影響
        if (r.needHeal) {
            await strategy.heal(r.token)
        }

        //延遲再重試
        await delay(5000)

    }

    //check
    if (errTemp !== null) {
        return Promise.reject(errToMsg(errTemp))
    }

    return b64
}


//coreAlive, 常駐模式: 出圖後瀏覽器留存供同行程後續呼叫重用, 含引用計數, 結束時重設閒置關閉排程
async function coreAlive(width, height, scale, html, opt = {}) {
    refCount++
    try {
        let b64 = await core(width, height, scale, html, opt, aliveStrategy)
        return b64
    }
    finally {
        refCount--
        scheduleIdleClose()
    }
}


//coreSingle, 單次模式: 每次啟動新瀏覽器出圖後即關閉釋放, Node行程可自然結束
async function coreSingle(width, height, scale, html, opt = {}) {
    let b64 = await core(width, height, scale, html, opt, singleStrategy)
    return b64
}


/**
 * 呼叫Chromium轉Html為png圖
 *
 * 提供兩種瀏覽器使用模式(opt.mode):
 * 'single'為每次呼叫啟動新瀏覽器, 出圖後即關閉釋放, Node行程可自然結束, 適合排程批次;
 * 'alive'為常駐瀏覽器, 第一次呼叫時啟動, 同行程後續呼叫重用故出圖較快, 閒置超過1小時自動關閉,
 * 批次腳本結束前可呼叫WHtml2png.close()主動關閉常駐瀏覽器, 使Node行程可立即退出(close前已開始之呼叫不再重試重啟瀏覽器);
 * 只有瀏覽器層錯誤(無法開頁、瀏覽器斷線)才於重試前重啟常駐瀏覽器, 頁面層錯誤不影響同時進行之其他呼叫
 *
 * 未指定opt.executablePath時使用套件自帶之免安裝chrome(只支援Windows), 經WHtml2png.ensureBins(即src/ensureBins.mjs)取得檔案組chrome:
 * 首次調用時下載(約200MB, 或自鏡像W_BIN_MIRROR取得)portable.zip之8個分割, 組合、解出portable/App/Chrome-bin/後逐檔以bin/manifest.json之SHA-256與大小校驗, 整組放入快取(預設本套件根目錄之bin-cache/<組識別>/, 與當前工作路徑無關),
 * 之後自快取取得(每行程首次亦校驗, 一檔不符即整組重取); 安裝套件時不下載; 環境變數W_BIN_CACHE、W_BIN_MIRROR、W_BIN_OFFLINE見README〈外部執行檔〉;
 * 取得chrome永不寫入當前工作路徑(僅opt.fdHtml/opt.fdErr之預設為當前工作路徑下之./_convertTemp)
 *
 * 頁面模板預先載入lodash(全域_)與wsemi(全域w), 皆由本套件相依之本機檔案供應(不需網路), 可於execJsHead、execJsPost使用
 *
 * 內容皆逐字插入頁面(含$&、$'、$$等字樣): html置於div#pl內; execJsHead與execJsPost以inline之script插入, 內容不得含字面</script>(需要時寫成<\/script>);
 * stylesHead與scriptsHead之網址置於屬性內, 不得含雙引號;
 * 未給opt.funGetUrl時以setContent載入(about:blank), html內之本機檔案(file://)與相對路徑資源不會載入, 請改用data url或給opt.funGetUrl
 *
 * @class
 * @param {Number} [width=700] 輸入圖片原始寬度數字，單位px，預設700
 * @param {Number} [height=400] 輸入圖片原始高度數字，單位px，預設400
 * @param {Number} [scale=3] 輸入欲將圖片放大比例數字，單位px，預設3
 * @param {String} html 輸入HTML字串，須為有效字串
 * @param {Object} [opt={}] 輸入設定物件，預設{}
 * @param {String} [opt.mode='single'] 輸入瀏覽器使用模式字串，'single'代表每次呼叫啟閉瀏覽器，'alive'代表常駐瀏覽器供同行程重用，預設'single'
 * @param {String} [opt.executablePath=''] 輸入瀏覽器執行檔路徑字串，可指定本機安裝Chrome或playwright託管chromium等(給予時不限作業系統)，未給則使用套件自帶之免安裝chrome(只支援Windows，經ensureBins取得，快取無檔時自動下載或自鏡像取得)，給予時不取套件自帶之chrome；注意alive模式之常駐瀏覽器由同行程第一次呼叫決定執行檔與modeHeadless，欲切換須先呼叫WHtml2png.close()，預設''

 * @param {Function} [opt.funPageWait=null] 輸入前端瀏覽器內偵測等待完成之函數，可用window或document等，回傳true則代表渲染完成可進行截圖，逾時60秒，預設null

 * @param {Integer} [opt.retry=3] 輸入總嘗試次數整數(含第一次)，每次失敗後等待5秒再試，預設3

 * @param {String} [opt.fdErr='./_convertTemp'] 輸入臨時儲存錯誤檔之資料夾位置字串(相對於當前工作路徑)，預設'./_convertTemp'
 * @returns {Promise} 回傳Promise，resolve為回傳base64圖片，reject為錯誤訊息字串(取不到套件自帶之chrome時為'[錯誤碼] 說明'，錯誤碼如E_OFFLINE_MISSING、E_HASH_MISMATCH)

 *             </div>
 *         </div>
 *         <div style="padding: 20px; border-top: 1px solid rgb(221, 221, 221); background-color: rgb(250, 250, 250); border-radius: 0px 0px 5px 5px;">
 *             Here is a panel footer
 *         </div>
 *     </div>
 * </div>
 *     `
 *     let width = 620
 *     let height = 235
 *     let scale = 3
 *
 *     let b64 = await WHtml2png(width, height, scale, html)
 *     // console.log('b64', b64)
 *
 *     // fs.writeFileSync('./test-scla.b64', b64, 'utf8')
 *     fs.writeFileSync('./test-scla.png', b64, { encoding: 'base64' })
 *
 *     console.log('finish')
 * }
 * testa()
 *     .catch((err) => {
 *         console.log(err)
 *     })
 *
 */
async function WHtml2png(width = 700, height = 400, scale = 3, html = '', opt = {}) {
    // console.log('WHtml2png', width, height, scale, opt)

    //mode
    let mode = get(opt, 'mode')
    if (mode !== 'single' && mode !== 'alive') {
        mode = 'single'
    }

    //dispatch
    if (mode === 'alive') {
        return coreAlive(width, height, scale, html, opt)
            .catch((err) => {
                return Promise.reject(errToMsg(err))
            })
    }
    return coreSingle(width, height, scale, html, opt)
        .catch((err) => {
            return Promise.reject(errToMsg(err))
        })
}


//close, 主動關閉alive模式之常駐瀏覽器並清除閒置計時器, 供批次腳本結束前呼叫使Node行程可立即退出; close前已開始之呼叫不再重試重啟瀏覽器
WHtml2png.close = async () => {
    closeEpoch++
    clearTimeout(timerIdle)
    timerIdle = null
    await closeGen(gen)
}
